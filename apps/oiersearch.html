#include #include using namespace std; const long long MOD = 998244353; // 快速幂求逆元 long long power(long long base, long long exp) { long long res = 1; base %= MOD; while (exp > 0) { if (exp % 2 == 1) res = (res * base) % MOD; base = (base * base) % MOD; exp /= 2; } return res; } long long modInverse(long long n) { return power(n, MOD - 2); } int main() { int n; cin >> n; // n <= 2 时无法满足条件 if (n <= 2) { cout << 0 << endl; return 0; } int MAX = 2 * n; vector fac(MAX + 1), invFac(MAX + 1); // 预处理阶乘和逆元 fac[0] = 1; invFac[0] = 1; for (int i = 1; i <= MAX; i++) { fac[i] = (fac[i - 1] * i) % MOD; } invFac[MAX] = modInverse(fac[MAX]); for (int i = MAX - 1; i >= 1; i--) { invFac[i] = (invFac[i + 1] * (i + 1)) % MOD; } long long U_n = 0; // 容斥原理计算 U_n for (int k = 0; k <= n; k++) { long long term; if (k == 0) { term = fac[n]; } else { term = 2LL * n * fac[2 * n - k - 1] % MOD; term = (term * invFac[k]) % MOD; term = (term * invFac[2 * n - 2 * k]) % MOD; term = (term * fac[n - k]) % MOD; } // 奇减偶加 if (k % 2 == 1) { U_n = (U_n - term + MOD) % MOD; } else { U_n = (U_n + term) % MOD; } } // 最终乘上女士的圆排列方案数 (n-1)! long long ans = (U_n * fac[n - 1]) % MOD; /* 注：如果题目明确认为座位是固定的（即座位1和座位2是不同的，不将旋转等价视为同一种）， 那么最后需乘上 2n，代码变为：ans = (ans * 2LL * n) % MOD; */ cout << ans << endl; return 0; }
